## 四、异常处理与日志

健壮爬虫的标配——异常处理让爬虫不会因一个失败就崩掉，日志让问题可追溯：

```python
import logging, time, random
import requests

logging.basicConfig(filename="crawler.log", level=logging.INFO)

def fetch(url, headers, max_retries=3):
    """带重试的请求函数"""
    for attempt in range(max_retries):
        try:
            resp = requests.get(url, headers=headers, timeout=5)
            resp.raise_for_status()
            return resp.text
        except requests.exceptions.RequestException:
            wait = 2 ** attempt + random.uniform(0, 1)
            logging.warning(f"第 {attempt + 1} 次请求失败,等 {wait:.1f}s 重试")
            time.sleep(wait)
    logging.error(f"请求 {url} 重试 {max_retries} 次后仍然失败")
    return None
```

> **日志级别**：`DEBUG`（调试信息）、`INFO`（一般信息）、`WARNING`（警告）、`ERROR`（错误）。爬虫中用 `WARNING` 和 `ERROR` 就够了。

## 六、完整爬虫架构模板

```python
"""
图书榜爬虫 - 完整版模板
包含：请求、解析、存储、异常处理、日志、控频
"""

import requests
from bs4 import BeautifulSoup
import csv
import sqlite3
import time
import logging
import random

logging.basicConfig(filename="crawler.log", level=logging.WARNING)

def fetch_page(url, headers):
    """请求一页,带重试和异常处理"""
    for attempt in range(3):
        try:
            resp = requests.get(url, headers=headers, timeout=5)
            resp.raise_for_status()
            return resp.text
        except requests.exceptions.RequestException as e:
            logging.warning(f"失败: {e}")
            time.sleep(random.uniform(1, 3))
    return None

def parse_page(html):
    """解析一页,返回记录列表"""
    soup = BeautifulSoup(html, "lxml")
    records = []
    for li in soup.select("li.book"):
        records.append({
            "title": li.select_one(".title").get_text(strip=True),
            "price": li.select_one(".price").get_text(strip=True),
        })
    return records

def save_csv(records, filename="books.csv"):
    """保存为 CSV"""
    with open(filename, "w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=["title", "price"])
        writer.writeheader()
        writer.writerows(records)
    logging.info(f"已保存 {filename}")

def save_sqlite(records, filename="books.db"):
    """保存为 SQLite"""
    conn = sqlite3.connect(filename)
    c = conn.cursor()
    c.execute("CREATE TABLE IF NOT EXISTS book (title TEXT, price TEXT)")
    c.executemany("INSERT INTO book VALUES (?, ?)",
                   [(r["title"], r["price"]) for r in records])
    conn.commit()
    conn.close()
    logging.info(f"已保存 {filename}")

def main():
    headers = {"User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64)..."}
    all_records = []
    
    # 翻页
    for page in range(1, 4):
        url = f"https://books.example.com/list?page={page}"
        html = fetch_page(url, headers)
        if not html:
            continue
        
        records = parse_page(html)
        all_records.extend(records)
        logging.info(f"第 {page} 页抓到 {len(records)} 条")
        
        time.sleep(random.uniform(1, 3))
    
    # 存储
    save_csv(all_records)
    save_sqlite(all_records)
    logging.info(f"全部完成,共 {len(all_records)} 条")

if __name__ == "__main__":
    main()
```
